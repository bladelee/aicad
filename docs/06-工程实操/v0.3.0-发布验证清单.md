## L1. 镜像可用性（2 分钟）

```bash
# 1. 登录（私有分发，PAT 需 read:packages）
echo <PAT> | docker login ghcr.io -u bladelee --password-stdin

# 2. 拉取
docker pull ghcr.io/bladelee/floorplan-mcp:v0.3.0
docker pull ghcr.io/bladelee/libredwg-cli:v0.3.0

# 3. CLI 入口冒烟
docker run --rm ghcr.io/bladelee/floorplan-mcp:v0.3.0 --version
# 期望: floorplan 0.3.0（随 tag 对齐）

docker run --rm ghcr.io/bladelee/floorplan-mcp:v0.3.0 --help
# 期望: 列出 7 个子命令

# 4. 镜像内 11 个 user_tools 注册验证（v0.3 核心修复点）
docker run --rm --entrypoint /opt/FreeCAD/usr/bin/python \
  -e PYTHONHOME=/opt/FreeCAD/usr -e LD_LIBRARY_PATH=/opt/FreeCAD/usr/lib -e HOME=/root \
  ghcr.io/bladelee/floorplan-mcp:v0.3.0 -c "
import sys
sys.path.insert(0, '/opt/floorplan'); sys.path.insert(0, '/opt/freecad-ai')
from mcp_server_stdio import link_user_tools
link_user_tools()
from freecad_ai.extensions.user_tools import load_user_tools
from freecad_ai.config import USER_TOOLS_DIR
names = sorted(t.name for t in load_user_tools(USER_TOOLS_DIR))
print('LOADED:', len(names)); print(names)
assert len(names) == 11, f'期望 11 个，实际 {len(names)} 个！'
print('L1-4 PASS')
" 2>/dev/null
# 期望: LOADED: 11 [...]  L1-4 PASS
```

## L3. 服务器形态 E2E（5 分钟，v0.3 新增能力）

```bash
TOKEN=$(openssl rand -hex 24)
PORT=3222
docker rm -f fp-l3 >/dev/null 2>&1

# 1. 起服务
docker run -d --name fp-l3 -p $PORT:3000 -v "$PWD/fp_l3:/data" \
  -e FLOORPLAN_AUTH_TOKEN=$TOKEN \
  ghcr.io/bladelee/floorplan-mcp:v0.3.0 serve-http

# 2. 等就绪
for i in $(seq 1 20); do curl -s -o /dev/null --max-time 2 "http://localhost:$PORT/sse" && break; sleep 2; done

# 3. 鉴权三连
code1=$(curl -s -o /dev/null -w "%{http_code}" "http://localhost:$PORT/sse")
code2=$(curl -s -o /dev/null -w "%{http_code}" -H "Authorization: Bearer wrong" "http://localhost:$PORT/sse")
ok=$(curl -s -N --max-time 4 -H "Authorization: Bearer $TOKEN" "http://localhost:$PORT/sse" | head -2)
echo "无token=$code1(期望401) 错token=$code2(期望401) 对token含endpoint=$(echo "$ok" | grep -c endpoint)(期望1)"
[ "$code1" = 401 ] && [ "$code2" = 401 ] && echo "$ok" | grep -q endpoint && echo "L3 PASS"

# 4. 工具面真实调用（SSE 会话 + tools/list）—— 用 freecad-ai 自带 client
docker exec fp-l3 /opt/FreeCAD/usr/bin/python -c "
import sys
sys.path.insert(0, '/opt/freecad-ai')
from freecad_ai.mcp.client import MCPClient
c = MCPClient('http://localhost:3000/sse', headers={'Authorization': 'Bearer $TOKEN'})
tools = c.list_tools()
ours = [t for t in tools if t.name.startswith('user_')]
print('TOOLS:', len(tools), '| OURS:', len(ours))
assert len(ours) == 11
print('L3-4 PASS: 远程 tools/list 11 个装修工具可见')
" 2>/dev/null

docker rm -f fp-l3 >/dev/null; rm -rf fp_l3
```

## L4. 真实客户场景（人工，10 分钟）

用真实项目图（`workdir/19-102/dxf/3-19-102平面系统图.dxf`）：

1. **DWG→DXF**（若客户给的是 DWG）：
   ```bash
   docker run --rm -v "$PWD:/data" ghcr.io/bladelee/libredwg-cli:v0.3.0 -y -o /data/out.dxf /data/xxx.dwg
   ```
2. **CLI 改墙+标注**（真实图 E6037）：
   ```bash
   docker run --rm -v "$PWD:/data" ghcr.io/bladelee/floorplan-mcp:v0.3.0 \
     move-wall-with-dim --dxf "/data/workdir/19-102/dxf/3-19-102平面系统图.dxf" \
     --handle E6037 --dx 500 --out /data/e2e_dim.dxf 2>/dev/null | grep -E '"ok"|synced_dim_count'
   # 期望: synced_dim_count: 1（DIMENSION 14A97E）
   ```
3. **Goose 服务器形态实测**：按 `floorplan-mcp/clients/goose-remote-http.md` 配好，说
   > "把 E6037 那堵墙向右移 500mm"
   验证 AI 调 `user_move_wall` 返回 JSON，服务器 `~/dwgs` 出现 `_moved.dxf`
4. **AutoCAD 校验**：客户打开输出 DXF，确认墙位置、DIMENSION 数值已更新（路径 C 的最终闭环，需客户参与）
